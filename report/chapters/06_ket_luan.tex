\chapter{Kết luận và hướng phát triển}
\label{ch:conclusion}

\section{Kết quả đạt được}
\begin{itemize}
  \item \textbf{Phân tích:} bóc tách đề bài thành 14 yêu cầu chức năng và 8 quy tắc nghiệp vụ có mã truy vết; làm rõ các điểm đề bài chưa quy định (sĩ số tối đa lớp online, ý nghĩa của “tách lớp”) bằng giả định tường minh.
  \item \textbf{Thiết kế:} 21 biểu đồ UML/ERD dạng mã nguồn (6 use case, 4 hoạt động, 4 tuần tự, 3 trạng thái, 2 lớp, 1 ERD, 1 kiến trúc); CSDL 14 bảng chuẩn 3NF với ràng buộc nghiệp vụ được cưỡng chế ngay trong CSDL.
  \item \textbf{Cài đặt:} ứng dụng web hoàn chỉnh gồm cổng học viên (4 chức năng bắt buộc + thanh toán + kiểm tra đầu vào) và dashboard văn phòng cho 3 vai trò; chat thời gian thực có hàng đợi, khôi phục phiên, chỉ báo đang soạn tin; dashboard thống kê trực quan.
  \item \textbf{Chất lượng:} 78 ca kiểm thử tự động đạt 100\%, bao phủ toàn bộ quy tắc sĩ số ở các giá trị biên và các luồng nghiệp vụ đầu – cuối.
  \item \textbf{Phát hiện trong phân tích:} chia đôi lớp chỉ hợp lệ khi tổng sĩ số $\geq 40$; hệ thống tự đề xuất phương án phù hợp (chia đều, tách phần dư, giữ ngoại lệ theo sức chứa phòng) giúp Lãnh đạo quyết định nhanh và đúng.
\end{itemize}

\section{Hạn chế}
\begin{itemize}
  \item Cổng thanh toán và việc gửi thông báo (email/SMS) mới ở mức mô phỏng.
  \item Chưa có chức năng điểm danh, nhập điểm từng buổi và đánh giá cuối khoá để “lưu trữ toàn bộ quá trình đào tạo” ở mức chi tiết.
  \item Ngân hàng câu hỏi kiểm tra đầu vào còn nhỏ (45 câu) và mới có dạng trắc nghiệm; chưa kiểm tra kỹ năng nghe, nói, viết.
  \item SQLite phù hợp quy mô một trung tâm; khi mở rộng nhiều cơ sở cần chuyển sang hệ quản trị CSDL máy chủ. Bộ đếm chống dò mật khẩu lưu trong bộ nhớ nên chỉ đúng với một tiến trình máy chủ.
  \item Token đăng nhập lưu ở \texttt{localStorage} của trình duyệt; bản triển khai thật nên dùng cookie \texttt{HttpOnly} và HTTPS.
\end{itemize}

\section{Hướng phát triển}
\begin{itemize}
  \item Tích hợp cổng thanh toán thật (VNPay/MoMo sandbox) với xác nhận qua IPN, gửi email/Zalo thông báo kết quả thi và lịch học.
  \item Bổ sung phân hệ đào tạo: điểm danh theo buổi, nhập điểm giữa kỳ – cuối kỳ, cảnh báo học viên vắng nhiều.
  \item Bài kiểm tra thích ứng (adaptive test) chọn câu theo năng lực, bổ sung phần nghe có audio.
  \item Hỗ trợ chatbot trả lời tự động các câu hỏi thường gặp ngoài giờ làm việc, chuyển sang tư vấn viên khi cần.
  \item Triển khai bằng Docker, chuyển CSDL sang PostgreSQL; bổ sung nhật ký kiểm toán (audit log) cho các quyết định của Lãnh đạo.
\end{itemize}
